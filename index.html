const canPlaceShape = (
    shape: Shape,
    startRow: number,
    startCol: number
  ): boolean => {
    for (let r = 0; r < shape.length; r++) {
      for (let c = 0; c < shape[r].length; c++) {
        if (shape[r][c] === 1) {
          const boardRow = startRow + r;
          const boardCol = startCol + c;
          if (
            boardRow >= BOARD_SIZE ||
            boardCol >= BOARD_SIZE
          ) {
            return false;
          }
          if (!isCellEmpty(boardRow, boardCol)) {
            return false;
          }
        }
      }
    }
    return true;
  };
